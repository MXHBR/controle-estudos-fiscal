// LER DADOS DO SUPABASE async carregarDadosDaNuvem() { try { this.syncStatus = '⏳ Carregando...'; const { data, error } = await supabaseClient .from('dados_usuario') .select('conteudo') .eq('id', 'dados_auditor') .maybeSingle(); if (error) throw error; if (data && data.conteudo) { const payload = typeof data.conteudo === 'string' ? JSON.parse(data.conteudo) : data.conteudo; if (payload.lembreteFixado !== undefined) this.lembreteFixado = payload.lembreteFixado; if (payload.materias) this.materias = payload.materias; if (payload.topicosPorMateria) this.topicosPorMateria = payload.topicosPorMateria; if (payload.questoesRegistro) this.questoesRegistro = payload.questoesRegistro; if (payload.cronogramaSemanas) this.cronogramaSemanas = payload.cronogramaSemanas; } else { // Se não existir o registro no banco, cria o inicial await this.salvarTudoNaNuvem(); } this.syncStatus = '☁️ Nuvem Sincronizada'; } catch (err) { console.error('Erro ao buscar dados no Supabase:', err); this.syncStatus = '❌ Erro ao Sincronizar'; } }, // GRAVAR DADOS NO SUPABASE (Com conversão segura contra Proxies do Alpine) async salvarTudoNaNuvem() { try { this.syncStatus = '⏳ Guardando...'; // JSON.parse(JSON.stringify(...)) remove todas as travas e proxies do Alpine.js const payloadSeguro = JSON.parse(JSON.stringify({ lembreteFixado: this.lembreteFixado, materias: this.materias, topicosPorMateria: this.topicosPorMateria, questoesRegistro: this.questoesRegistro, cronogramaSemanas: this.cronogramaSemanas })); const { error } = await supabaseClient .from('dados_usuario') .upsert({ id: 'dados_auditor', conteudo: payloadSeguro, atualizado_em: new Date().toISOString() }); if (error) throw error; this.syncStatus = '☁️ Nuvem Sincronizada'; } catch (err) { console.error('Erro ao gravar no Supabase:', err); this.syncStatus = '❌ Erro ao Sincronizar'; } }
